import socket
import pickle
import select
import sys
import time

serverHostName  = str(sys.argv[1])
serverPort = int(sys.argv[2])
fileName = str(sys.argv[3])
chunkSize = int(sys.argv[4])

sc = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
sc.bind(('localhost', 0))

file = open(fileName, 'wb')

offset = 0
block_size = chunkSize
t0 = time.time()

def waitForReply( uSocket ):
    rx, tx, er = select.select( [uSocket], [], [], 1)
    # waits for data or timeout after 1 second
    if rx==[]:
        return False
    else:
        return True

while True:
    request = (fileName, offset, block_size)
    req = pickle.dumps(request)
    sc.sendto(req, (serverHostName, serverPort))

    while not waitForReply(sc):
        sc.sendto(req, (serverHostName, serverPort))

    data, server_address = sc.recvfrom(block_size*4)

    if not data:
        continue

    reply = pickle.loads(data)
    if reply[0] == 1:
        print("File not found.")
        break
    if reply[0] == 2:
        print("Offset out of bounds.")
        break
    dataLen = reply[1]
    dataReceived = reply[2]    
    
    if len(data) < block_size:
        print("Reached EOF. Writing final chunk.")
        file.write(dataReceived)
        break

    offset += chunkSize
    print("Updated offset:", offset)

    file.write(dataReceived)

t1 = time.time()
F_size = offset  # Assuming the offset represents the total file size in bytes
Tr = F_size / (t1 - t0)

# Convert the transfer rate to Kbytes per second (KB/s)
Tr_KBps = Tr / 1024

duration = t1 - t0
minutes, seconds = divmod(duration, 60)

print("Duration: {} minutes {} seconds".format(int(minutes), int(seconds)))
print("Transfer Rate: {:.2f} KB/s".format(Tr_KBps))
